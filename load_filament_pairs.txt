def load_filament_pairs(manifest_path: str) -> list[tuple[str, str]]:
    """Load a pair manifest, converting COCO annotations when necessary."""
    manifest_file = Path(manifest_path)
    with manifest_file.open(encoding="utf-8") as file:
        manifest = json.load(file)

    if isinstance(manifest, list):
        return manifest
    if not isinstance(manifest, dict):
        raise ValueError(f"Unsupported manifest format in {manifest_file}")

    images = manifest.get("images")
    annotations = manifest.get("annotations")
    if not isinstance(images, list) or not isinstance(annotations, list):
        raise ValueError(f"Manifest is missing COCO images or annotations: {manifest_file}")

    image_root = manifest_file.parent / "train_images"
    mask_root = manifest_file.parent.parent.parent / "fil_modified" / "images"
    mask_root.mkdir(parents=True, exist_ok=True)
    annotations_by_image = defaultdict(list)
    for annotation in annotations:
        annotations_by_image[annotation["image_id"]].append(annotation)

    pairs = []
    for image_info in images:
        image_path = image_root / Path(image_info["file_name"]).name
        mask_path = mask_root / f"{image_info['id']}.png"
        if not image_path.is_file():
            raise FileNotFoundError(f"Training image not found: {image_path}")
        if not mask_path.is_file():
            width, height = image_info["width"], image_info["height"]
            mask = Image.new("L", (width, height), 0)
            draw = ImageDraw.Draw(mask)
            for annotation in annotations_by_image.get(image_info["id"], []):
                segmentation = annotation.get("segmentation", [])
                if segmentation and isinstance(segmentation[0], (int, float)):
                    segmentation = [segmentation]
                for polygon in segmentation:
                    if len(polygon) < 6:
                        continue
                    points = list(zip(polygon[0::2], polygon[1::2]))
                    draw.polygon(points, fill=255)
            mask.save(mask_path)
        pairs.append((str(image_path), str(mask_path)))

    return pairs
